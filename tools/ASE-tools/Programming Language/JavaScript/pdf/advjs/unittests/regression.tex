test('adds 1 + 2 to equal 3', 
  () => {
    expect(add(1, 2)).toBe(3);
});
test('adds 1 + 2 to equal 3', 
  () => {
    expect(sub(1, 2)).toBe(-1);
});

\begin{frame}[fragile]{Regression Test}

\begin{itemize}
	\item When A or B makes a change, they need to run all the tests to ensure the change does not break any existing code. 
	\item In this case, A or B executes the command `jest' to run all the tests in the project.
	\item Or, they can run `npm test' to run the same command. 
\end{itemize}
