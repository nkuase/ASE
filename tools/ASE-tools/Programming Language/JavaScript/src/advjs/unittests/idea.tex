\begin{frame}[fragile]{UnitTest Example in Java}

\begin{itemize}
	\item This example shows how we can check the module (Calculator class) provides the functionality (multiply) we expect. 
	\item The CalculatorTest UnitTest checks if $4 \times 5 == 20$ in Java (JUnit Framework). 
\end{itemize}

\begin{columns}%
\begin{column}{0.5\textwidth}%
\begin{Code}{}%
\begin{lstlisting}[firstnumber=1, label=glabels, xleftmargin=0pt, basicstyle=\tiny]% 

public class Calculator {
  public int multiply(int a, int b) {
    return a * b;
  }
}
\end{lstlisting}%   
\end{Code}%
\end{column}%
\begin{column}{0.5\textwidth}%

\begin{Code}{}%
\begin{lstlisting}[firstnumber=1, label=glabels, xleftmargin=0pt, basicstyle=\tiny]% 

class CalculatorTest {
  Calculator calculator;
  @BeforeEach                                         
  void setUp() {
      calculator = new Calculator();
  }
  @Test                                               
  @DisplayName("Simple")
  void testMultiply() {
    assertEquals(20, 
      calculator.multiply(4, 5),     
      "Regular multiplication should work");  
  }
}  
\end{lstlisting}%   
\end{Code}%

\end{column}%
\end{columns}%
